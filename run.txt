# 1. Start PostgreSQL
run Docker Desktop
docker compose up -d


# 2. Create and activate virtual environment
python -m venv .venv
.venv\Scripts\activate      # Windows
# source .venv/bin/activate # macOS / Linux

# 3. Install dependencies and load data
pip install -r requirements.txt
python load.py

# 4. Run Postgres
docker exec -it designing_sql-pg-1 psql -U myuser -d datapack


# 5. Run queries
python run_queries.py


# 6. Initialize NPX app
npx create-next-app@latest web --typescript --tailwind --eslint --app --src-dir --import-alias "@/*"
cd web
npm install pg @types/pg


# 7. Create file .env.example and copy it into .env.local.
echo "DATABASE_URL=postgresql://myuser:mypassword@127.0.0.1:54321/datapack" > .env.example.

copy .env.example .env.local.

# 8. Run app
npm run dev


# 9. Open browser:
http://localhost:3000/companies
